The model talks to SQLite through src/sqlite_db.f90, a small iso_c_binding interface to libsqlite3. Create the database, tables and content ahead of time; the app opens it read-only at startup.

Base your model on src/marsupials.f90. Always use prepared statements with bound parameters and finalize them when you are done:

subroutine find_by_name(db, query, names, ok)
  type(database), intent(inout) :: db
  character(len=*), intent(in) :: query
  type(kv_pair), allocatable, intent(out) :: names(:)
  logical, intent(out) :: ok
  type(statement) :: stmt
  integer :: rc

  ok = .false.
  if (db%prepare('SELECT name, latinName FROM marsupials WHERE name = ?1', stmt) /= SQLITE_OK) return
  rc = stmt%bind_text(1, query)       ! never concatenate user input into SQL

  allocate(names(0))
  do
    rc = stmt%step()
    if (rc /= SQLITE_ROW) exit
    names = [names, kv_pair(stmt%column_text(0), stmt%column_text(1))]
  end do
  call stmt%finalize()
  ok = rc == SQLITE_DONE
end subroutine find_by_name
