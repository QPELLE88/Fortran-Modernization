The model talks to SQLite through src/database.f90, a thin ISO_C_BINDING wrapper over libsqlite3. You should create the database, tables, and content ahead of time.

Base your model on src/marsupial.f90. Use prepared statements and bind user input as a parameter, never concatenate it into SQL:

subroutine find_marsupial(query, found, item, ok)
  character(len=*), intent(in) :: query
  logical, intent(out) :: found, ok
  type(marsupial_t), intent(out) :: item
  type(database_t) :: db
  type(statement_t) :: stmt
  integer :: rc

  found = .false.
  call db%open('marsupials.sqlite3', ok, SQLITE_OPEN_READONLY)
  if (.not. ok) return

  call db%prepare('SELECT name, latinName, wikiLink, description FROM marsupials ' // &
                  'WHERE INSTR(LOWER(name), LOWER(?1)) > 0 LIMIT 1', stmt, rc)
  if (rc == SQLITE_OK) rc = stmt%bind_text(1, trim(query))
  if (rc == SQLITE_OK) then
    if (stmt%step() == SQLITE_ROW) then
      found = .true.
      item%name = stmt%column_text(0)
      item%latin_name = stmt%column_text(1)
      item%wiki_link = stmt%column_text(2)
      item%description = stmt%column_text(3)
    end if
  end if
  ok = rc == SQLITE_OK

  call stmt%finalize()
  call db%close()
end subroutine find_marsupial
