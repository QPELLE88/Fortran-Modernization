The model uses FLIBS Fortran module and C files to talk to SQLite. You should create the database, tables, and content ahead of time.

You should base your model on src/marsupial.f90 to make sure it includes the right features. Rows are returned as a derived type with allocatable strings, so there is no fixed column width or row limit:

type :: marsupial_t
  character(len=:), allocatable :: name, latin_name, wiki_link, description
end type marsupial_t

subroutine find_marsupial(query, found, m)
  character(len=*), intent(in)  :: query
  logical, intent(out)          :: found
  type(marsupial_t), intent(out) :: m

  type(marsupial_t), allocatable :: rows(:)

  ! sql_quote escapes the user-provided string before it reaches SQLite
  call select_marsupials( &
    'WHERE INSTR(LOWER(name), LOWER(' // sql_quote(trim(query)) // ')) LIMIT 1', rows)

  found = size(rows) > 0
  if (found) m = rows(1)
end subroutine find_marsupial
